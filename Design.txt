`timescale 1ns/1ps

module axi4lite_to_ahblite_bridge #(
    parameter ADDR_WIDTH = 32,
    parameter DATA_WIDTH = 32
)(
    input logic ACLK,
    input logic ARESETn,

    input  logic [ADDR_WIDTH-1:0] AWADDR,
    input  logic                  AWVALID,
    output logic                  AWREADY,

    input  logic [DATA_WIDTH-1:0] WDATA,
    input  logic [DATA_WIDTH/8-1:0] WSTRB,
    input  logic                  WVALID,
    output logic                  WREADY,

    output logic [1:0] BRESP,
    output logic       BVALID,
    input  logic       BREADY,

    input  logic [ADDR_WIDTH-1:0] ARADDR,
    input  logic                  ARVALID,
    output logic                  ARREADY,

    output logic [DATA_WIDTH-1:0] RDATA,
    output logic [1:0]            RRESP,
    output logic                  RVALID,
    input  logic                  RREADY,

    input logic HCLK,
    input logic HRESETn,

    output logic [ADDR_WIDTH-1:0] HADDR,
    output logic                  HWRITE,
    output logic [1:0]            HTRANS,
    output logic [2:0]            HSIZE,
    output logic [2:0]            HBURST,

    output logic [DATA_WIDTH-1:0] HWDATA,

    input logic [DATA_WIDTH-1:0] HRDATA,
    input logic                  HREADY,
    input logic                  HRESP
);

    localparam [1:0] HTRANS_IDLE   = 2'b00;
    localparam [1:0] HTRANS_NONSEQ = 2'b10;

    localparam [1:0] AXI_OKAY      = 2'b00;
    localparam [1:0] AXI_SLVERR    = 2'b10;

    typedef enum logic [2:0] {
        IDLE,
        AHB_WRITE_DATA,
        AXI_WRITE_RESP,
        AHB_READ_DATA
    } state_t;

    state_t state;

    logic [ADDR_WIDTH-1:0] saved_addr;
    logic [DATA_WIDTH-1:0] saved_wdata;

    assign HSIZE  = 3'b010;
    assign HBURST = 3'b000;
    assign RDATA  = HRDATA;

    always_comb begin
        AWREADY = (state == IDLE) && !BVALID && !RVALID;
        WREADY  = (state == IDLE) && !BVALID && !RVALID;
        ARREADY = (state == IDLE) && !BVALID && !RVALID;
    end

    always_ff @(posedge HCLK or negedge HRESETn) begin
        if (!HRESETn || !ARESETn) begin
            state       <= IDLE;
            saved_addr  <= '0;
            saved_wdata <= '0;
            BRESP       <= AXI_OKAY;
            BVALID      <= 1'b0;
            RRESP       <= AXI_OKAY;
            RVALID      <= 1'b0;
            HADDR       <= '0;
            HWRITE      <= 1'b0;
            HTRANS      <= HTRANS_IDLE;
            HWDATA      <= '0;
        end else begin
            if (BVALID && BREADY) BVALID <= 1'b0;
            if (RVALID && RREADY) RVALID <= 1'b0;

            case (state)
                IDLE: begin
                    HTRANS <= HTRANS_IDLE;
                    if (AWVALID && WVALID) begin
                        saved_addr  <= AWADDR;
                        saved_wdata <= WDATA;
                        HADDR       <= AWADDR;
                        HWRITE      <= 1'b1;
                        HTRANS      <= HTRANS_NONSEQ;
                        HWDATA      <= WDATA;
                        state       <= AHB_WRITE_DATA;
                    end else if (ARVALID) begin
                        saved_addr  <= ARADDR;
                        HADDR       <= ARADDR;
                        HWRITE      <= 1'b0;
                        HTRANS      <= HTRANS_NONSEQ;
                        state       <= AHB_READ_DATA;
                    end
                end

                AHB_WRITE_DATA: begin
                    HTRANS <= HTRANS_IDLE;
                    if (HREADY) begin
                        BRESP  <= HRESP ? AXI_SLVERR : AXI_OKAY;
                        BVALID <= 1'b1;
                        state  <= AXI_WRITE_RESP;
                    end
                end

                AXI_WRITE_RESP: begin
                    if (BREADY) begin
                        state <= IDLE;
                    end
                end

                AHB_READ_DATA: begin
                    HTRANS <= HTRANS_IDLE;
                    if (HREADY) begin
                        RRESP  <= HRESP ? AXI_SLVERR : AXI_OKAY;
                        RVALID <= 1'b1;
                        state  <= IDLE;
                    end
                end

                default: state <= IDLE;
            endcase
        end
    end

endmodule
